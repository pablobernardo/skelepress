% templates/book-novel.tex
%
% Builtin book style "book-novel" (skelepress book --template book-novel or
% pandoc.template: book-novel). Classic literary typography for fiction on A4:
% serif body with generous leading, wide reading margins, centered chapter
% openings, understated running headers. Select it by bare name; the shipped
% `book` style and consumer-owned templates are unaffected.
%
% Self-contained on purpose: the TeX engine runs in the render directory with
% no TEXINPUTS setup, so builtin templates cannot reliably \input a shared
% file. Shared behavior lives in the repeated blocks below (engine detection,
% pandoc compatibility, the bookkit BookChapters* hooks and the opt-in
% running-headers section), mirroring templates/book.tex.
%
% Supported contract variables mirror book.tex: document.* title page fields,
% toc, babel-lang, cover-image, titlepage, running-headers, highlighting-macros.

\documentclass[11pt,openany]{book}

% ---- Engine detection (mirrors templates/book.tex) ----
\newif\ifxetexorlua
\ifdefined\luatexversion
    \xetexorluatrue
\else
    \ifdefined\XeTeXversion
        \xetexorluatrue
    \else
        \xetexorluafalse
    \fi
\fi

\ifxetexorlua
    \usepackage{fontspec}
    % Literary serif first, then the toolkit's fallback chain.
    \IfFontExistsTF{EB Garamond}{
        \setmainfont{EB Garamond}
    }{
        \IfFontExistsTF{TeX Gyre Pagella}{
            \setmainfont{TeX Gyre Pagella}
        }{
            \IfFontExistsTF{Times New Roman}{
                \setmainfont{Times New Roman}
            }{
                \setmainfont{Latin Modern Roman}
            }
        }
    }
\else
    \usepackage[T1]{fontenc}
    \usepackage[utf8]{inputenc}
    \usepackage{lmodern}
\fi

% ---- Language ----
\usepackage[$if(babel-lang)$$babel-lang$$else$english$endif$]{babel}

% ---- Page geometry: generous reading margins ----
\usepackage[a4paper,top=2.8cm,bottom=3.2cm,inner=2.8cm,outer=2.2cm]{geometry}
\setlength{\emergencystretch}{3em}
\linespread{1.12}

% ---- Headers/footers: quiet, page number centered at the bottom ----
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\chaptermark}[1]{\markboth{#1}{}}
\renewcommand{\sectionmark}[1]{\markright{#1}}
\fancyhead[LE,RO]{\small\itshape\nouppercase{\leftmark}}
\fancyfoot[C]{\small\thepage}
\renewcommand{\headrulewidth}{0pt}

% ---- Graphics ----
\usepackage{graphicx}
\usepackage{etoolbox}
\makeatletter
\def\maxwidth{\ifdim\Gin@nat@width>\linewidth 0.90\linewidth\else\Gin@nat@width\fi}
\def\maxheight{\ifdim\Gin@nat@height>0.75\textheight 0.75\textheight\else\Gin@nat@height\fi}
\makeatother
\setkeys{Gin}{width=\maxwidth, height=\maxheight, keepaspectratio}

% ---- Code blocks: toolkit look, quieter frame for prose-heavy books ----
\usepackage{fancyvrb}
\usepackage{fvextra}
\RecustomVerbatimEnvironment{verbatim}{Verbatim}{}
\usepackage{framed}
\fvset{fontsize=\small}
\usepackage{mdframed}
\usepackage{xcolor}
\definecolor{codegray}{gray}{0.96}
\BeforeBeginEnvironment{Highlighting}{%
    \begin{mdframed}[
    backgroundcolor=codegray,
    roundcorner=3pt,
    leftmargin=0pt,
    rightmargin=0pt,
    innerleftmargin=6pt,
    innerrightmargin=6pt,
    innertopmargin=4pt,
    innerbottommargin=4pt,
    linewidth=0pt
    ]}
\AfterEndEnvironment{Highlighting}{\end{mdframed}}
$if(highlighting-macros)$
$highlighting-macros$
$endif$
\makeatletter
\@ifundefined{Shaded}{
    \definecolor{shadecolor}{gray}{0.95}
    \newenvironment{Shaded}{\begin{snugshade}}{\end{snugshade}}
}{}
\@ifundefined{Highlighting}{
    \DefineVerbatimEnvironment{Highlighting}{Verbatim}{commandchars=\\\{\}}
}{}
\makeatother
\fvset{breaklines=true,breakanywhere=false,breakafter={/.,=}}

% ---- Tables ----
\usepackage{longtable,booktabs,array}
\usepackage{calc}
% Pandoc uses this counter for captionless tables instead of stepping table.
\ifcsname c@none\endcsname\else\newcounter{none}\fi

% ---- Pandoc compatibility helpers ----
$if(strikeout)$
% Older Pandoc writers emit \sout; newer ones emit \st.
\usepackage[normalem]{ulem}
\let\st\sout
$endif$
\providecommand{\tightlist}{%
    \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\pandocbounded}[1]{%
    \begin{minipage}{\linewidth}#1\end{minipage}}

% ---- Chapter presentation: centered opening with a small ornament ----
\usepackage{titlesec}
\newcommand{\NovelChapterOrnament}{%
    \vspace{0.2em}{\large\textasteriskcentered}\vspace{0.4em}}
\titleformat{\chapter}[display]
    {\centering\normalfont}
    {\scshape\chaptername~\thechapter}
    {6pt}
    {\normalfont\huge\itshape}
    [\vspace{4pt}\NovelChapterOrnament]
\titlespacing*{\chapter}{0pt}{-1em}{2em}
\titleformat{\section}
    {\normalfont\large\itshape}{\thesection}{1em}{}
\titleformat{\subsection}
    {\normalfont\normalsize\itshape}{\thesubsection}{1em}{}

% ---- Links ----
\usepackage[hidelinks]{hyperref}

% ---- Contents entries the adapter can hide (mirrors templates/book.tex) ----
\makeatletter
\newif\ifBookChapterListed
\BookChapterListedtrue
\let\BookAddContentsLine\addcontentsline
\renewcommand\addcontentsline[3]{%
  \ifBookChapterListed\BookAddContentsLine{#1}{#2}{#3}\fi}
\newcommand\BookChaptersListed{\BookChapterListedtrue}
\newcommand\BookChaptersUnlisted{\BookChapterListedfalse}
\makeatother

% ---- Title page: classic literary layout ----
\newcommand{\doTitlePage}{%
\begin{titlepage}
\centering
\vspace*{3cm}
{\Huge\itshape $if(title)$$title$$endif$\par}
$if(subtitle)$
    \vspace{0.6cm}{\Large $subtitle$\par}
$endif$
\vspace{1.6cm}
{\large\textasteriskcentered\par}
\vfill
$if(author)$
    {\large $author$\par}
$endif$
$if(publisher)$
    \vspace{0.8cm}{\small\scshape $publisher$\par}
$endif$
$if(date)$
    \vspace{0.2cm}{\small $date$\par}
$endif$
$if(rights)$
    \vspace{0.6cm}{\footnotesize $rights$\par}
$endif$
\end{titlepage}
\pagenumbering{gobble}
\clearpage
\pagenumbering{arabic}
}

% ---- Book-level mode commands (mirrors templates/book.tex) ----
\makeatletter
\renewcommand\frontmatter{\clearpage\@mainmatterfalse\pagenumbering{roman}}
\renewcommand\mainmatter{\clearpage\@mainmattertrue\pagenumbering{arabic}}
\makeatother

$if(running-headers)$
% ---- Opt-in running headers (mirrors templates/book.tex) ----
\usepackage[breakall,fit]{truncate}
\fancyhead{}
\fancyhead[L]{\fontsize{8}{10}\selectfont\itshape\truncate{0.46\textwidth}{$if(title)$$title$$endif$}}
\fancyhead[R]{\fontsize{8}{10}\selectfont\itshape\truncate{0.46\textwidth}{\nouppercase{\leftmark}}}
\renewcommand{\sectionmark}[1]{}
\let\RunningHeaderChapter\chapter
\RenewDocumentCommand{\chapter}{s o m}{%
  \IfBooleanTF{#1}{\RunningHeaderChapter*{#3}}{%
    \IfNoValueTF{#2}{\RunningHeaderChapter{#3}}{\RunningHeaderChapter[#2]{#3}}}%
  \IfNoValueTF{#2}{\markboth{#3}{}}{\markboth{#2}{}}}
\patchcmd{\cleardoublepage}{\hbox{}}{\hbox{}\thispagestyle{empty}}{}{}
$endif$

% ---- Document ----
\begin{document}

$if(cover-image)$
\newgeometry{margin=0pt}
\thispagestyle{empty}
\noindent\includegraphics[width=\paperwidth,height=\paperheight,keepaspectratio]{$cover-image$}
\restoregeometry
$endif$

$if(titlepage)$
    \doTitlePage
$endif$

\frontmatter

$if(toc)$
\tableofcontents
\clearpage
$if(running-headers)$
\markboth{}{}
$endif$
$endif$

$body$

\end{document}
